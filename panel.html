/******************************************************* * ProLink — API PANEL TÉCNICO (solo JSON, sin pantalla) * Este proyecto de Apps Script NO sirve HTML. Solo responde * JSON a pedidos que llegan desde panel.html (alojado en * GitHub Pages) vía fetch(). Así el técnico nunca visita una * URL de script.google.com y no ve el aviso de Google. *******************************************************/ // ==================== CONFIGURACIÓN ==================== const CONFIG = { SHEET_ID: 'PEGAR_AQUI_EL_ID_DE_TU_GOOGLE_SHEET', SHEETS: { RECLAMOS: 'Reclamos_Clientes', CATALOGO: 'Catalogo_Precios_y_Servicios', REPARACIONES: 'Registro_Reparaciones', COMPROBANTES: 'Comprobantes', CATALOGO_OBRAS: 'Catalogo_Obras_Mantenimiento', TAREAS_INTERNAS: 'Tareas_Internas', REGISTRO_TAREAS_INTERNAS: 'Registro_Tareas_Internas' }, NOMBRE_EMPRESA: 'ProLink', CARPETA_COMPROBANTES: 'Comprobantes_ProLink', LOGO_FILE_ID: 'PEGAR_AQUI_EL_ID_DEL_LOGO_EN_DRIVE', // Token simple para que no cualquiera que encuentre la URL pueda escribir datos. // Cambialo por algo propio, y poné el mismo valor en panel.html (const API_TOKEN). // Ojo: como panel.html es un archivo público en GitHub, esto NO es seguridad fuerte, // solo frena bots/curiosos casuales. Si el repo es privado, es más que suficiente. API_TOKEN: 'cambiar-este-token' }; function ss_() { return SpreadsheetApp.openById(CONFIG.SHEET_ID); } function sheet_(name) { return ss_().getSheetByName(name); } function esFilaBasura_(row) { const primera = row[0]; const resto = row.slice(1); const restoVacio = resto.every(v => v === '' || v === null || v === undefined); const primeraVacia = primera === '' || primera === null || primera === undefined; if (primeraVacia && restoVacio) return true; if (!primeraVacia && restoVacio) return true; return false; } function generarId_(prefijo, sheet) { const ultimaFila = sheet.getLastRow(); const numero = ultimaFila < 2 ? 1 : ultimaFila; const ts = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyMMdd'); return `${prefijo}-${ts}-${String(numero).padStart(4, '0')}`; } // ==================== LOGO ==================== function logoBlob_() { return DriveApp.getFileById(CONFIG.LOGO_FILE_ID).getBlob(); } function logoDataUri() { const cache = CacheService.getScriptCache(); const cacheado = cache.get('LOGO_DATA_URI'); if (cacheado) return cacheado; const blob = logoBlob_(); const base64 = Utilities.base64Encode(blob.getBytes()); const dataUri = `data:${blob.getContentType()};base64,${base64}`; cache.put('LOGO_DATA_URI', dataUri, 21600); return dataUri; } // ==================== CATÁLOGOS ==================== function obtenerCatalogo() { const sh = sheet_(CONFIG.SHEETS.CATALOGO); const data = sh.getDataRange().getValues(); data.shift(); return data .filter(r => !esFilaBasura_(r) && r[0]) .map(r => ({ codigo: r[0], descripcion: r[1], categoria: r[2], precioCliente: r[3], comisionTecnico: r[4] })); } function obtenerCatalogoObras() { const sh = sheet_(CONFIG.SHEETS.CATALOGO_OBRAS); const data = sh.getDataRange().getValues(); data.shift(); return data .filter(r => !esFilaBasura_(r) && r[0]) .map(r => ({ codigo: r[0], descripcion: r[1], categoria: r[2], unidad: r[3], precioUnitario: r[4] })); } // ==================== RECLAMOS DE CLIENTES ==================== function obtenerTareas(filtros) { filtros = filtros || {}; const sh = sheet_(CONFIG.SHEETS.RECLAMOS); const data = sh.getDataRange().getValues(); data.shift(); return data .map((r, i) => ({ fila: i + 2, raw: r })) .filter(x => !esFilaBasura_(x.raw) && x.raw[0]) .map(x => { const r = x.raw; return { fila: x.fila, id: r[0], fecha: r[1], nombre: r[2], dni: r[3], direccion: r[4], barrio: r[5], telefono: r[6], tipoFalla: r[7], descripcion: r[8], ubicacion: r[9], estado: r[10], tecnico: r[11], zona: r[12], prioridad: r[13] || 'Verde' }; }) .filter(t => { if (filtros.estado && filtros.estado !== 'Todos' && t.estado !== filtros.estado) return false; if (filtros.zona && filtros.zona !== 'Todas' && t.zona !== filtros.zona) return false; if (filtros.prioridad && filtros.prioridad !== 'Todas' && t.prioridad !== filtros.prioridad) return false; if (filtros.busqueda) { const b = filtros.busqueda.toLowerCase(); if (!(`${t.nombre} ${t.direccion} ${t.id}`.toLowerCase().includes(b))) return false; } return true; }); } function asignarPrioridad(fila, prioridad, esInterna) { const sh = esInterna ? sheet_(CONFIG.SHEETS.TAREAS_INTERNAS) : sheet_(CONFIG.SHEETS.RECLAMOS); const columna = esInterna ? 7 : 14; sh.getRange(fila, columna).setValue(prioridad); return { ok: true }; } function asignarTecnico(fila, tecnico) { const sh = sheet_(CONFIG.SHEETS.RECLAMOS); sh.getRange(fila, 12).setValue(tecnico); sh.getRange(fila, 11).setValue('En proceso'); return { ok: true }; } function cerrarTrabajo(data) { const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const shRec = sheet_(CONFIG.SHEETS.RECLAMOS); const estadoActual = shRec.getRange(data.filaReclamo, 11).getValue(); const ESTADOS_YA_CERRADOS = ['Resuelto', 'Requiere Segunda Visita', 'Cancelado']; if (ESTADOS_YA_CERRADOS.includes(estadoActual)) { // Ya se cerró antes (doble click / doble tap). No generamos otro comprobante. throw new Error('YA_CERRADO:' + estadoActual); } const shRep = sheet_(CONFIG.SHEETS.REPARACIONES); const idRep = generarId_('REP', shRep); const fecha = new Date(); const montoCobrado = data.tipoFallaConfirmado === 'Externa' ? 0 : data.articulos.reduce((sum, a) => sum + (a.precioCliente * a.cantidad), 0); const manoObra = data.articulos.reduce((sum, a) => sum + (a.comisionTecnico * a.cantidad), 0); const articulosTexto = data.articulos.map(a => `${a.codigo} x${a.cantidad}`).join(', '); shRep.appendRow([ idRep, data.idReclamo, fecha, data.tecnico, data.diagnostico, data.tipoFallaConfirmado, articulosTexto, data.materiales || '', montoCobrado, data.metodoPago, manoObra, data.estadoFinal, '', '' ]); shRec.getRange(data.filaReclamo, 11).setValue(data.estadoFinal); return { ok: true, id: idRep, montoCobrado, manoObra, filaReparacion: shRep.getLastRow() }; } finally { lock.releaseLock(); } } // ==================== OBRAS Y MANTENIMIENTO ==================== function obtenerTareasInternas(filtros) { filtros = filtros || {}; const sh = sheet_(CONFIG.SHEETS.TAREAS_INTERNAS); const data = sh.getDataRange().getValues(); data.shift(); return data .map((r, i) => ({ fila: i + 2, raw: r })) .filter(x => !esFilaBasura_(x.raw) && x.raw[0]) .map(x => { const r = x.raw; return { fila: x.fila, id: r[0], fecha: r[1], tipo: r[2], titulo: r[3], descripcion: r[4], zona: r[5], prioridad: r[6] || 'Verde', estado: r[7], tecnico: r[8], fechaProgramada: r[9] }; }) .filter(o => { if (filtros.tipo && filtros.tipo !== 'Todos' && o.tipo !== filtros.tipo) return false; if (filtros.estado && filtros.estado !== 'Todos' && o.estado !== filtros.estado) return false; if (filtros.prioridad && filtros.prioridad !== 'Todas' && o.prioridad !== filtros.prioridad) return false; if (filtros.busqueda) { const b = filtros.busqueda.toLowerCase(); if (!(`${o.titulo} ${o.zona} ${o.id}`.toLowerCase().includes(b))) return false; } return true; }); } function cerrarTareaInterna(data) { const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const shTareas = sheet_(CONFIG.SHEETS.TAREAS_INTERNAS); const estadoActual = shTareas.getRange(data.filaTarea, 8).getValue(); const ESTADOS_YA_CERRADOS = ['Resuelto', 'Requiere Continuidad', 'Cancelado']; if (ESTADOS_YA_CERRADOS.includes(estadoActual)) { throw new Error('YA_CERRADO:' + estadoActual); } const shReg = sheet_(CONFIG.SHEETS.REGISTRO_TAREAS_INTERNAS); const idCierre = generarId_('CINT', shReg); const fecha = new Date(); const montoTotal = data.items.reduce((sum, it) => sum + (it.precioUnitario * it.cantidad), 0); const itemsTexto = data.items.map(it => `${it.codigo} x${it.cantidad}`).join(', '); shReg.appendRow([idCierre, data.idTarea, fecha, data.tecnico, data.trabajoRealizado, itemsTexto, data.materiales || '', montoTotal, data.estadoFinal]); shTareas.getRange(data.filaTarea, 8).setValue(data.estadoFinal); return { ok: true, id: idCierre, montoTotal: montoTotal }; } finally { lock.releaseLock(); } } // ==================== COMPROBANTES PDF ==================== function generarNumeroComprobante_() { const lock = LockService.getScriptLock(); lock.waitLock(30000); try { const props = PropertiesService.getScriptProperties(); let ultimo = Number(props.getProperty('ULTIMO_COMPROBANTE') || '0'); ultimo += 1; props.setProperty('ULTIMO_COMPROBANTE', String(ultimo)); return String(ultimo).padStart(6, '0'); } finally { lock.releaseLock(); } } function carpetaComprobantes_() { const nombre = CONFIG.CARPETA_COMPROBANTES; const carpetas = DriveApp.getFoldersByName(nombre); return carpetas.hasNext() ? carpetas.next() : DriveApp.createFolder(nombre); } function generarPdfComprobante_(datos) { const doc = DocumentApp.create(`Comprobante_${datos.numero}_temp`); const body = doc.getBody(); body.clear(); body.setMarginTop(30).setMarginBottom(30).setMarginLeft(40).setMarginRight(40); const imagen = body.appendImage(logoBlob_()); const anchoOriginal = imagen.getWidth(); const altoOriginal = imagen.getHeight(); const anchoFinal = 160; imagen.setWidth(anchoFinal); imagen.setHeight(Math.round(altoOriginal * (anchoFinal / anchoOriginal))); body.appendParagraph(' '); body.appendParagraph('COMPROBANTE DE PAGO').setHeading(DocumentApp.ParagraphHeading.HEADING1).setAlignment(DocumentApp.HorizontalAlignment.CENTER); body.appendParagraph(`N° ${datos.numero}`).setAlignment(DocumentApp.HorizontalAlignment.CENTER).setBold(true); body.appendParagraph(' '); const fechaTexto = Utilities.formatDate(datos.fecha, Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm"); body.appendParagraph(`Fecha: ${fechaTexto}`); body.appendParagraph(`Cliente: ${datos.cliente}`); body.appendParagraph(`Dirección: ${datos.direccion}`); body.appendParagraph(' '); const tabla = body.appendTable([['Ítem', 'Monto'], ['Visita técnica', `$${datos.monto}`]]); tabla.getRow(0).editAsText().setBold(true); body.appendParagraph(' '); body.appendParagraph(`TOTAL ABONADO: $${datos.monto}`).setBold(true); body.appendParagraph(' '); body.appendParagraph(`Técnico: ${datos.tecnico}`); body.appendParagraph(' '); body.appendParagraph(`${CONFIG.NOMBRE_EMPRESA} - Comprobante interno de pago por visita técnica.`).setItalic(true); doc.saveAndClose(); const archivoDoc = DriveApp.getFileById(doc.getId()); const pdfBlob = archivoDoc.getAs('application/pdf'); const carpeta = carpetaComprobantes_(); const pdfFile = carpeta.createFile(pdfBlob).setName(`Comprobante_${datos.numero}.pdf`); archivoDoc.setTrashed(true); return { base64: Utilities.base64Encode(pdfBlob.getBytes()), fileUrl: pdfFile.getUrl() }; } function emitirComprobante(data) { const numero = generarNumeroComprobante_(); const fecha = new Date(); const pdf = generarPdfComprobante_({ numero: numero, cliente: data.cliente, direccion: data.direccion, monto: data.monto, tecnico: data.tecnico, fecha: fecha }); const shComp = sheet_(CONFIG.SHEETS.COMPROBANTES); shComp.appendRow([numero, fecha, data.idReparacion, data.idReclamo, data.cliente, data.direccion, 'Visita técnica', data.monto, data.tecnico, pdf.fileUrl]); if (data.filaReparacion) { const shRep = sheet_(CONFIG.SHEETS.REPARACIONES); shRep.getRange(data.filaReparacion, 13).setValue('Sí'); shRep.getRange(data.filaReparacion, 14).setValue(numero); } return { ok: true, numero: numero, pdfBase64: pdf.base64, fileUrl: pdf.fileUrl }; } // ==================== ROUTER JSON (llamado desde panel.html vía fetch) ==================== /** * Acciones de solo lectura -> se pueden llamar por GET o POST. * Acciones que escriben datos -> exigen el token en el payload. */ const ACCIONES_LECTURA = ['obtenerCatalogo', 'obtenerCatalogoObras', 'obtenerTareas', 'obtenerTareasInternas', 'logoDataUri']; const ACCIONES_ESCRITURA = ['asignarPrioridad', 'asignarTecnico', 'cerrarTrabajo', 'cerrarTareaInterna', 'emitirComprobante']; function ejecutarAccion_(action, payload) { payload = payload || {}; const esEscritura = ACCIONES_ESCRITURA.includes(action); if (esEscritura && payload.token !== CONFIG.API_TOKEN) { throw new Error('Token inválido.'); } switch (action) { case 'obtenerCatalogo': return obtenerCatalogo(); case 'obtenerCatalogoObras': return obtenerCatalogoObras(); case 'obtenerTareas': return obtenerTareas(payload.filtros); case 'obtenerTareasInternas': return obtenerTareasInternas(payload.filtros); case 'logoDataUri': return logoDataUri(); case 'asignarPrioridad': return asignarPrioridad(payload.fila, payload.prioridad, payload.esInterna); case 'asignarTecnico': return asignarTecnico(payload.fila, payload.tecnico); case 'cerrarTrabajo': return cerrarTrabajo(payload.data); case 'cerrarTareaInterna': return cerrarTareaInterna(payload.data); case 'emitirComprobante': return emitirComprobante(payload.data); default: throw new Error('Acción desconocida: ' + action); } } function jsonOutput_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function doGet(e) { try { const action = e.parameter.action; if (!action) return jsonOutput_({ ok: true, mensaje: 'API Panel Técnico ProLink activa.' }); const payload = e.parameter.payload ? JSON.parse(e.parameter.payload) : {}; const data = ejecutarAccion_(action, payload); return jsonOutput_({ ok: true, data: data }); } catch (err) { return jsonOutput_({ ok: false, error: err.message }); } } function doPost(e) { try { const body = JSON.parse(e.postData.contents); const data = ejecutarAccion_(body.action, body.payload); return jsonOutput_({ ok: true, data: data }); } catch (err) { return jsonOutput_({ ok: false, error: err.message }); } }
